// ---------- Пусни веднъж (и след промяна в листовете): листове, заглавия, примерни кодове, ключ за таблото ----------
function setup() {
  const ss = book_();
  ss.setSpreadsheetTimeZone(TZ);
  if (/^(Untitled|Неозаглавена)/i.test(ss.getName())) ss.rename(SHOP + ' — магазин');
  Object.keys(SHEETS).forEach(function (name) {
    const sh = ss.getSheetByName(name) || ss.insertSheet(name);
    if (sh.getLastRow() === 0) {
      sh.appendRow(SHEETS[name]);
      sh.setFrozenRows(1);
    } else {
      // по-стара таблица: липсващите колони отиват най-вдясно
      const head = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(String);
      SHEETS[name].forEach(function (h) { if (head.indexOf(h) < 0) sh.getRange(1, sh.getLastColumn() + 1).setValue(h); });
    }
    sh.getRange(1, 1, 1, sh.getLastColumn()).setFontWeight('bold');
  });
  ['Sheet1', 'Лист1', 'Лист 1'].forEach(function (n) {
    const s = ss.getSheetByName(n);
    if (s && s.getLastRow() === 0) ss.deleteSheet(s);
  });
  listRule_('Поръчки', 'Статус', STATUSES);
  listRule_('Запитвания', 'Статус', LEAD_STATUSES);
  const codes = ss.getSheetByName('Кодове');
  codes.getRange(2, 5, codes.getMaxRows() - 1, 2).setNumberFormat('@');   // датите като текст: 2026-12-31
  if (codes.getLastRow() === 1) {
    codes.appendRow(['SLOY10', 'процент', 10, 0, '', '2026-12-31', 100, 0, 'да']);
    codes.appendRow(['HALLOWEEN', 'сума', 3, 20, '', '2026-10-31', '', 0, 'да']);
  }
  // „Склад“ остава празен, докато не го попълниш — дотогава сайтът предлага всички цветове
  const props = PropertiesService.getScriptProperties();
  if (!props.getProperty('ADMIN_KEY')) props.setProperty('ADMIN_KEY', randomKey_());
  Logger.log('Имейлите отиват на: ' + owner_() + ' (остават ' + MailApp.getRemainingDailyQuota() + ' за днес)');
  showKey();
}

// ---------- Вход ----------
function doGet(e) {
  const p = (e && e.parameter) || {};
  try {
    if (p.action === 'code') return json_(checkCode_(p.code));
    if (p.action === 'stock') return json_({ ok: true, stock: stockMap_(), card: !!stripeKey_() });
    return json_({ ok: true, shop: SHOP });
  } catch (err) {
    return json_({ ok: false, error: msg_(err) });
  }
}

function doPost(e) {
  try {
    const d = JSON.parse(e.postData.contents);
    switch (d.action || 'files') {
      case 'files': return json_(saveFiles_(d));
      case 'order': return json_(addOrder_(d));
      case 'lead': return json_(addLead_(d));
      case 'useCode': return json_(useCode_(d));
      case 'payCheck': return json_(payCheck_(d, false));
      case 'login': auth_(d.key); return json_({ ok: true });
      case 'data': auth_(d.key); return json_(data_());
      case 'status': auth_(d.key); return json_(setStatus_(d));
      case 'notify': auth_(d.key); return json_(notify_(d));
      case 'leadStatus': auth_(d.key); return json_(setLead_(d));
      case 'saveStock': auth_(d.key); return json_(saveStock_(d.rows));
      case 'saveCodes': auth_(d.key); return json_(saveCodes_(d.rows));
      case 'econt': auth_(d.key); return json_(econtLabel_(d));
      case 'econtDelete': auth_(d.key); return json_(econtDelete_(d));
      case 'econtStatus': auth_(d.key); return json_(econtStatus_(d));
      case 'payRefresh': auth_(d.key); return json_(payCheck_(d, true));
      case 'payLink': auth_(d.key); return json_(payLink_(d));
      case 'markPaid': auth_(d.key); return json_(markPaid_(d));
    }
    return json_({ ok: false, error: 'unknown action' });
  } catch (err) {
    return json_({ ok: false, error: msg_(err) });
  }
}

// ---------- Файлове → Drive ----------
function saveFiles_(d) {
  const files = (d.files || []).slice(0, MAX_FILES);
  if (!files.length) throw new Error('Няма файлове');
  throttle_('files', 40, 600);
  const folder = getFolder_(ROOT_FOLDER).createFolder(String(d.order || 'без номер').slice(0, 60));
  const saved = files.map(function (f) {
    const name = String(f.name || 'file').replace(/[\\/:*?"<>|]+/g, '_').slice(0, 100);
    const ext = (name.match(/\.([a-z0-9]+)$/i) || [])[1];
    const type = ext && FILE_TYPES[ext.toLowerCase()];
    if (!type) throw new Error('Неразрешен тип файл: ' + name);
    const bytes = Utilities.base64Decode(f.data);
    if (bytes.length > MAX_MB * 1024 * 1024) throw new Error('Файлът е над ' + MAX_MB + ' MB: ' + name);
    const file = folder.createFile(Utilities.newBlob(bytes, type, name));
    return { name: f.name, url: file.getUrl() };
  });
  return { ok: true, folder: folder.getUrl(), files: saved };
}

// fields = полетата за имейла (на български, както ги праща сайтът); customer = редовете за клиента на неговия език
function addOrder_(d) {
  const f = d.fields || {};
  const c = d.customer || {};
  if (!str_(f['Име']) || !str_(f['Телефон'])) throw new Error('Липсват име или телефон');
  throttle_('orders', 30, 600);
  const sent = String(d.order || f['Поръчка'] || '');
  let order = /^SL-\d{6}-\d{3,4}$/.test(sent) ? sent : '';
  const dup = withLock_(function () {
    const sh = sheet_('Поръчки');
    const col = cols_(sh);
    if (order) {
      const row = findRow_(sh, col['Поръчка'], order);
      if (row) {
        // същият телефон = същата поръчка, изпратена повторно; иначе — нов номер
        if (digits_(sh.getRange(row, col['Телефон']).getValue()) === digits_(f['Телефон'])) return true;
        order = '';
      }
    }
    if (!order) order = uniqueNo_(sh, col['Поръчка'], 'SL');
    appendObj_(sh, {
      'Дата': new Date(), 'Поръчка': order, 'Статус': STATUSES[0], 'Общо': f['Общо'], 'Сума €': Number(d.amount) || '',
      'Име': f['Име'], 'Телефон': f['Телефон'], 'Имейл': f['Имейл'], 'Доставка': f['Доставка'], 'Град': f['Град'],
      'Офис / адрес': f['Офис / адрес'], 'Плащане': f['Плащане'], 'Продукти': f['Продукти'] || f['Модели'],
      'Експресна': f['Експресна'], 'Отстъпка': f['Отстъпка'], 'Доставка за сметка на': f['Доставка за сметка на'],
      'Бележка': f['Бележка'], 'Език': d.lang === 'en' ? 'EN' : 'BG', 'Тема': d.subject,
      'Еконт офис': String((d.ship && d.ship.office) || '').slice(0, 20), 'Пощ. код': String((d.ship && d.ship.postCode) || '').slice(0, 10),
      'Филамент': useText_(d.use),
    });
    return false;
  });
  const bank = /превод/i.test(String(f['Плащане'] || '')) ? bankDetails_() : '';
  if (dup) return { ok: true, order: order, bank: bank };
  f['Поръчка'] = order;
  // плащане с карта / Apple Pay / Google Pay: страница на Stripe за плащане, клиентът отива там веднага
  let pay = '';
  if (/карта/i.test(String(f['Плащане'] || '')) && stripeKey_() && Number(d.amount) > 0) {
    try { pay = checkout_(order, Number(d.amount), email_(f['Имейл']), d.lang === 'en').url; } catch (err) { console.error('Stripe: ' + msg_(err)); }
  }
  const subject = oneLine_(String(d.subject || 'Нова поръчка ' + order).split(sent || '\u0000').join(order));
  try {
    mail_(owner_(), subject, fieldsText_(f) + footer_(), email_(f['Имейл']));
  } catch (err) { console.error('имейл до нас: ' + msg_(err)); }
  try {
    confirmOrder_(order, f, c, d.lang === 'en', pay);
  } catch (err) { console.error('потвърждение: ' + msg_(err)); }
  return { ok: true, order: order, bank: bank, pay: pay };
}

function setStatus_(d) {
  if (STATUSES.indexOf(d.status) < 0) throw new Error('Невалиден статус');
  return withLock_(function () {
    const sh = sheet_('Поръчки');
    const col = cols_(sh);
    const row = findRow_(sh, col['Поръчка'], String(d.order));
    if (!row) throw new Error('Няма такава поръчка');
    sh.getRange(row, col['Статус']).setValue(d.status);
    if (d.tracking !== undefined) sh.getRange(row, col['Товарителница']).setValue(clean_(String(d.tracking).slice(0, 60)));
    if (d.note !== undefined) sh.getRange(row, col['Моя бележка']).setValue(clean_(String(d.note).slice(0, 1000)));
    // пуснат печат → филаментът и консумативите излизат от склада (веднъж на поръчка)
    let used = '';
    if (d.status === 'Печата се' && col['От склада'] && !String(sh.getRange(row, col['От склада']).getValue()).trim()) {
      used = takeStock_(String(sh.getRange(row, col['Филамент']).getValue()));
      sh.getRange(row, col['От склада']).setValue(Utilities.formatDate(new Date(), TZ, 'dd.MM HH:mm') + ' · ' + used);
    }
    return { ok: true, used: used, stock: used ? stockRows_() : undefined };
  });
}

// ---------- Склад: изваждане при печат ----------
// d.use от сайта: [{ m: 'PLA', c: ['Черен'], g: 85, q: 1, k: { 'Халка с верижка за ключодържател': 1 } }]
// В листа се пази четимо: „PLA · Черен · 85 г × 1 [+ Халка с верижка за ключодържател × 1]; …“ — и се чете обратно оттук.
function useText_(use) {
  if (!Array.isArray(use)) return '';
  return use.slice(0, 50).map(function (u) {
    const cs = (u.c || []).map(function (c) { return String(c).slice(0, 30); });
    const k = Object.keys(u.k || {}).map(function (n) { return String(n).slice(0, 50) + ' × ' + Math.max(1, Math.round(Number(u.k[n]) || 1)); });
    return String(u.m || '').slice(0, 30) + ' · ' + cs.join(' + ') + ' · ' + Math.max(0, Math.round(Number(u.g) || 0)) + ' г × ' + Math.max(1, Math.round(Number(u.q) || 1))
      + (k.length ? ' [+ ' + k.join(', ') + ']' : '');
  }).join('; ');
}

function parseUse_(text) {
  return String(text || '').split('; ').map(function (part) {
    const m = part.match(/^(.+?) · (.+?) · (\d+) г × (\d+)(?: \[\+ (.+)\])?$/);
    if (!m) return null;
    const k = {};
    (m[5] || '').split(', ').forEach(function (x) { const y = x.match(/^(.+) × (\d+)$/); if (y) k[y[1]] = Number(y[2]); });
    return { m: m[1], c: m[2].split(' + '), g: Number(m[3]), q: Number(m[4]), k: k };
  }).filter(Boolean);
}

// вади грамовете (+брак) от ролките и консумативите; връща кратко описание; праща имейл, ако нещо свършва
function takeStock_(text) {
  const use = parseUse_(text);
  const sh = sheet_('Склад');
  const v = sh.getDataRange().getValues();
  const find = function (mat, color) {
    for (let i = 1; i < v.length; i++) if (String(v[i][0]).trim() === mat && String(v[i][1]).trim() === color) return i;
    return -1;
  };
  const done = [], missing = [];
  const take = function (mat, color, amount, unit) {
    const i = find(mat, color);
    if (i < 0) { missing.push(mat === CONSUMABLE ? color : mat + ' ' + color); return; }
    v[i][2] = Math.max(0, Math.round((Number(v[i][2]) || 0) - amount));
    sh.getRange(i + 1, 3).setValue(v[i][2]);
    done.push((mat === CONSUMABLE ? color : mat + ' ' + color) + ' −' + Math.round(amount) + ' ' + unit);
  };
  use.forEach(function (u) {
    const total = u.g * u.q * (1 + WASTE);
    u.c.forEach(function (c) { take(u.m, c, total / u.c.length, 'г'); });   // многоцветните — поравно
    Object.keys(u.k).forEach(function (n) { take(CONSUMABLE, n, u.k[n] * u.q, 'бр.'); });
  });
  if (use.length) take(CONSUMABLE, MAILER, 1, 'бр.');
  lowStockAlert_();
  if (!use.length) return 'няма данни за материала — извади ръчно';
  return done.join(', ') + (missing.length ? ' · няма в склада: ' + missing.join(', ') : '');
}

// ролки под LOW_GRAMS и консумативи под LOW_PCS → един имейл; „Сигнал“ пази кога е пратен, за да не идва всеки път
function lowStockAlert_() {
  const sh = sheet_('Склад');
  const v = sh.getDataRange().getValues();
  const low = [];
  for (let i = 1; i < v.length; i++) {
    if (!v[i][0] || !v[i][1]) continue;
    const pcs = String(v[i][0]).trim() === CONSUMABLE;
    const left = Number(v[i][2]) || 0;
    const isLow = left < (pcs ? LOW_PCS : LOW_GRAMS);
    if (isLow && !String(v[i][4] || '').trim()) {
      low.push((pcs ? String(v[i][1]) : v[i][0] + ' ' + v[i][1]) + ' — остават ' + left + (pcs ? ' бр.' : ' г'));
      sh.getRange(i + 1, 5).setValue(Utilities.formatDate(new Date(), TZ, 'dd.MM HH:mm'));
    }
    if (!isLow && String(v[i][4] || '').trim()) sh.getRange(i + 1, 5).setValue('');
  }
  if (low.length) {
    try {
      mail_(owner_(), 'Свършва на склад: ' + low.length + (low.length === 1 ? ' позиция' : ' позиции') + ' — ' + SHOP,
        'Време е да поръчаш:\n\n• ' + low.join('\n• ') + '\n\nПраговете: ролка под ' + LOW_GRAMS + ' г, консуматив под ' + LOW_PCS + ' бр.' + footer_());
    } catch (err) { console.error('сигнал за склад: ' + msg_(err)); }
  }
  return low;
}

// ---------- Плащане с карта: Stripe Checkout (карта, Apple Pay, Google Pay) ----------
// Ключът е в Script properties → STRIPE_SECRET (sk_test_… за проба, sk_live_… на живо) — въвежда го собственикът.
// Без него сайтът не показва плащането с карта.
function stripeKey_() {
  return String(PropertiesService.getScriptProperties().getProperty('STRIPE_SECRET') || '').trim();
}

function stripe_(method, path, params) {
  const key = stripeKey_();
  if (!key) throw new Error('Няма STRIPE_SECRET в Script properties');
  const opt = { method: method, muteHttpExceptions: true, headers: { Authorization: 'Bearer ' + key } };
  if (params) opt.payload = form_(params);
  const res = UrlFetchApp.fetch('https://api.stripe.com/v1/' + path, opt);
  let j;
  try { j = JSON.parse(res.getContentText('UTF-8')); } catch (e) { throw new Error('Stripe отговори с ' + res.getResponseCode()); }
  if (res.getResponseCode() >= 400) throw new Error('Stripe: ' + ((j.error && j.error.message) || res.getResponseCode()));
  return j;
}

// { a: { b: 1 }, c: [x] } → a[b]=1&c[0]=x (форматът на Stripe)
function form_(obj) {
  const out = [];
  (function walk(v, k) {
    if (v === undefined || v === null || v === '') return;
    if (typeof v === 'object') Object.keys(v).forEach(function (n) { walk(v[n], k ? k + '[' + n + ']' : n); });
    else out.push(encodeURIComponent(k) + '=' + encodeURIComponent(v));
  })(obj, '');
  return out.join('&');
}

// заявка към Еконт: с профила от Script properties — истинската система; без него — тестовата (demo.econt.com)
function econt_(path, body) {
  const auth = econtAuth_();
  const base = auth ? 'https://ee.econt.com/services/' : 'https://demo.econt.com/ee/services/';
  const res = UrlFetchApp.fetch(base + path, {
    method: 'post', contentType: 'application/json', payload: JSON.stringify(body), muteHttpExceptions: true,
    headers: { Authorization: 'Basic ' + Utilities.base64Encode(auth || 'iasp-dev:1Asp-dev', Utilities.Charset.UTF_8) },
  });
  const code = res.getResponseCode();
  const text = res.getContentText('UTF-8');
  let j;
  try { j = JSON.parse(text); } catch (e) { throw new Error('Еконт отговори с ' + code + (code === 401 ? ' — провери ECONT_USER и ECONT_PASS' : '')); }
  if (code >= 400 || (j && j.type && j.innerErrors)) throw new Error('Еконт: ' + econtError_(j));
  return j;
}

// грешките на Еконт са вложени — събираме смислените съобщения
function econtError_(e) {
  const out = [];
  (function walk(x) {
    if (!x) return;
    const m = String(x.message || '').trim();
    if (m && m !== ':' && out.indexOf(m) < 0) out.push(m);
    (x.innerErrors || []).forEach(walk);
  })(e);
  return out.join(' ') || 'неизвестна грешка';
}

// редовете на листа като обекти по заглавията; най-новите първо. _ts = датата в милисекунди (за статистиката)
function table_(name, limit) {
  const v = sheet_(name).getDataRange().getValues();
  const head = v[0].map(String);
  return v.slice(1).filter(function (r) { return r[1]; }).map(function (r) {
    const o = {};
    head.forEach(function (h, i) {
      const x = r[i];
      if (x instanceof Date) { o[h] = Utilities.formatDate(x, TZ, 'dd.MM.yyyy HH:mm'); if (i === 0) o._ts = x.getTime(); }
      else o[h] = typeof x === 'number' ? x : String(x);
    });
    return o;
  }).reverse().slice(0, limit);
}

  if (!code) return { ok: false, reason: 'empty' };
  const v = sheet_('Кодове').getDataRange().getValues();
  for (let i = 1; i < v.length; i++) {
    const r = v[i];
    if (String(r[0]).trim().toUpperCase() !== code) continue;
    if (r[8] === false || /^(не|no|false)$/i.test(String(r[8]).trim())) return { ok: false, reason: 'invalid' };
    const today = ymd_(new Date());
    const from = ymd_(r[4]), to = ymd_(r[5]);
    if (from && today < from) return { ok: false, reason: 'notyet' };
    if (to && today > to) return { ok: false, reason: 'expired' };
    const max = Number(r[6]) || 0, used = Number(r[7]) || 0;
    if (max && used >= max) return { ok: false, reason: 'used' };
    return {
      ok: true,
      type: /сума|fixed|€|eur/i.test(String(r[1])) ? 'fixed' : 'percent',
      value: Number(r[2]) || 0,
      minOrder: Number(r[3]) || 0,
    };
  }
  return { ok: false, reason: 'invalid' };
}

function useCode_(d) {
  const code = String(d.code || '').trim().toUpperCase();
  return withLock_(function () {
    const sh = sheet_('Кодове');
    const v = sh.getDataRange().getValues();
    for (let i = 1; i < v.length; i++) {
      if (String(v[i][0]).trim().toUpperCase() === code) {
        sh.getRange(i + 1, 8).setValue((Number(v[i][7]) || 0) + 1);
        return { ok: true };
      }
    }
    return { ok: false, reason: 'invalid' };
  });
}

function codeRows_() {
  const v = sheet_('Кодове').getDataRange().getValues();
  return v.slice(1).filter(function (r) { return String(r[0]).trim(); }).map(function (r) {
    return {
      code: String(r[0]).trim().toUpperCase(), type: /сума|fixed|€|eur/i.test(String(r[1])) ? 'сума' : 'процент',
      value: Number(r[2]) || 0, minOrder: Number(r[3]) || 0, from: ymd_(r[4]), to: ymd_(r[5]),
      max: Number(r[6]) || 0, used: Number(r[7]) || 0, active: !(r[8] === false || /^(не|no|false)$/i.test(String(r[8]).trim())),
    };
  });
}

function saveCodes_(rows) {
  if (!Array.isArray(rows)) throw new Error('Няма редове');
  const date = function (s) { s = String(s || '').trim(); if (s && !/^\d{4}-\d{2}-\d{2}$/.test(s)) throw new Error('Датата трябва да е като 2026-12-31: ' + s); return s; };
  const seen = {};
  const clean = rows.slice(0, 500).map(function (r) {
    const code = String(r.code || '').trim().toUpperCase();
    if (!/^[A-Z0-9][A-Z0-9_-]{1,29}$/.test(code)) throw new Error('Кодът е само с латински букви и цифри (2–30 знака): ' + (code || 'празен'));
    if (seen[code]) throw new Error('Кодът се повтаря: ' + code);
    seen[code] = true;
    const value = Number(r.value) || 0;
    if (value <= 0) throw new Error('Стойността на ' + code + ' трябва да е над 0');
    const type = r.type === 'сума' ? 'сума' : 'процент';
    if (type === 'процент' && value > 100) throw new Error('Процентът на ' + code + ' е над 100');
    return [code, type, value, Math.max(0, Number(r.minOrder) || 0), date(r.from), date(r.to),
      Number(r.max) > 0 ? Math.round(Number(r.max)) : '', Math.max(0, Math.round(Number(r.used) || 0)), r.active ? 'да' : 'не'];
  });
  return withLock_(function () {
    const sh = sheet_('Кодове');
    if (sh.getLastRow() > 1) sh.getRange(2, 1, sh.getLastRow() - 1, 9).clearContent();
    sh.getRange(2, 5, sh.getMaxRows() - 1, 2).setNumberFormat('@');
    if (clean.length) sh.getRange(2, 1, clean.length, 9).setValues(clean);
    return { ok: true, saved: clean.length };
  });
}

function adminKey_() {
  return PropertiesService.getScriptProperties().getProperty('ADMIN_KEY') || '';
}

function randomKey_() {
  const abc = 'abcdefghjkmnpqrstuvwxyz23456789';
  const bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, Utilities.getUuid() + Date.now());
  let s = '';
  for (let i = 0; i < 16; i++) s += abc[(bytes[i] + 256) % abc.length] + (i % 4 === 3 && i < 15 ? '-' : '');
  return s;
}

// ключът се сравнява; след 20 грешни опита входът спира за 15 минути
function auth_(key) {
  const real = adminKey_();
  if (!real) throw new Error('Няма ключ за таблото — пусни setup() в Apps Script');
  const cache = CacheService.getScriptCache();
  const fails = Number(cache.get('fails') || 0);
  if (fails >= 20) throw new Error('Твърде много грешни опити — изчакай 15 минути');
  if (String(key || '') !== real) {
    cache.put('fails', String(fails + 1), 900);
    throw new Error('Грешен ключ');
  }
}

// защита от залпове заявки: най-много limit пъти за seconds секунди (за целия сайт)
function throttle_(name, limit, seconds) {
  const cache = CacheService.getScriptCache();
  const n = Number(cache.get('t:' + name) || 0);
  if (n >= limit) throw new Error('Твърде много заявки — опитай пак след няколко минути');
  cache.put('t:' + name, String(n + 1), seconds);
}

// до 3 автоматични имейла на един адрес за 6 часа — някой да не ползва сайта, за да спами чужда поща
function allowMail_(to) {
  const cache = CacheService.getScriptCache();
  const k = 'm:' + to.toLowerCase().slice(0, 200);
  const n = Number(cache.get(k) || 0);
  if (n >= 3) return false;
  cache.put(k, String(n + 1), 21600);
  return true;
}

// полетата за имейла: „Етикет: стойност“, а многоредовите — на отделен блок
function fieldsText_(f) {
  return Object.keys(f).filter(function (k) { return f[k] !== undefined && f[k] !== ''; }).map(function (k) {
    const v = String(f[k]);
    return v.indexOf('\n') >= 0 ? '\n' + k + ':\n' + v + '\n' : k + ': ' + v;
  }).join('\n');
}

// номерата на колоните по заглавията (редът в листа може да се мести)
function cols_(sh) {
  const head = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
  const map = {};
  head.forEach(function (h, i) { map[String(h).trim()] = i + 1; });
  return map;
}

function appendObj_(sh, obj) {
  const col = cols_(sh);
  const row = [];
  for (let i = 0; i < sh.getLastColumn(); i++) row.push('');
  Object.keys(obj).forEach(function (k) { if (col[k]) row[col[k] - 1] = clean_(obj[k]); });
  sh.appendRow(row);
}

function listRule_(name, header, list) {
  const sh = sheet_(name);
  const c = cols_(sh)[header];
  sh.getRange(2, c, sh.getMaxRows() - 1, 1).setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(list, true).build());
}

function findRow_(sh, col, value) {
  const last = sh.getLastRow();
  if (last < 2 || !col) return 0;
  const v = sh.getRange(2, col, last - 1, 1).getValues();
  for (let i = 0; i < v.length; i++) if (String(v[i][0]) === value) return i + 2;
  return 0;
}

// SL-251015-123 / Z-251015-123 — ако номерът е зает, следващ случаен
function uniqueNo_(sh, col, prefix) {
  const day = Utilities.formatDate(new Date(), TZ, 'yyMMdd');
  for (let i = 0; i < 60; i++) {
    const n = prefix + '-' + day + '-' + (i < 30 ? 100 + Math.floor(Math.random() * 900) : 1000 + Math.floor(Math.random() * 9000));
    if (!findRow_(sh, col, n)) return n;
  }
  throw new Error('Няма свободен номер');
}

function withLock_(fn) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try { return fn(); } finally { lock.releaseLock(); }
}

// текст от сайта: без формули (=, +, -, @ в началото), телефоните остават текст (0888… не става 888…)
function clean_(v) {
  if (v === undefined || v === null) return '';
  if (v instanceof Date || typeof v === 'number') return v;
  const s = String(v).slice(0, 5000);
  return /^[=+\-@]/.test(s) || /^[\d\s]+$/.test(s) ? "'" + s : s;
}

function str_(v) {
  return String(v || '').trim();
}

function oneLine_(v) {
  return String(v || '').replace(/[\r\n]+/g, ' ').trim();
}

function digits_(v) {
  return String(v || '').replace(/\D/g, '');
}

function email_(v) {
  const s = String(v || '').trim();
  return /^[^\s@<>()",;]+@[^\s@<>()",;]+\.[a-z]{2,}$/i.test(s) && s.length < 200 ? s : '';
}

function ymd_(v) {
  if (!v) return '';
  if (v instanceof Date) return Utilities.formatDate(v, TZ, 'yyyy-MM-dd');
  return String(v).trim();
}

}
